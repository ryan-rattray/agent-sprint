from memory import SlidingWindow

MAX_ITERATIONS = 61 # Caps the loops to 10 iterations so a runaway model can't run on forever

TRACE = True # Set to False to run quietly. True prints the full trace every turn.

def show(label, text):
    """Print one labeled trace section, but only when TRACE is on."""
    if not TRACE:
        return
    print(label)
    print(text)
    print()

# The prompt is the contract, now shorter than week 1 as JSON needs less explaining.
# Also, the example is removed because JSON IS the example.
PROMPT = """You are a research agent. You answer questions by using tools, then cite your sources.
Reply with ONLY a JSON object, no other text. To use a tool:
{"tool": "<tool name>", "arguments": { ... }}
Available tools: run_command (arguments: command, a list), read_file (arguments: rel_path),
fetch_url (arguments: url, timeout).
Do not wrap your reply in markdown or code fences. Send the raw JSON object only.
Reply with exactly one JSON object and nothing else. Never write an Observation line. Observations come only from the tools, after your reply ends.
This computer runs Windows. There is no ls program. To list files, use run_command with ["cmd", "/c", "dir"].
When you have enough to answer, reply with ONLY:
{"answer": "<your answer>", "citations": ["<file or URL>", ...]}

Question: {question}
"""


def observation_from(result):
    """Wrap a tool's return dictionary as the Observation text. 
    Takes the tool's return and decide ok or error.
    """
    if result["ok"]:
        return f"Observation: {result.get('text') or result.get('stdout') or ''}"
    return f"Observation: tool error ({result['error']}): {result.get('detail', '')}"


def run(question, model=ask_gateway):
    """Runs through the whole loop.  
    - Ask the question.
    - Validate the response.
    - Dispatch a tool.
    - Append the observation.
    - Repeat until final answer.
    - Final answer dictionary:  {"answer": str, "citations": [str, ...]}."""

    prompt = PROMPT.replace("{question}", question)
    # NEW (Week 3, Day 1): the window owns this run's history
    window = SlidingWindow(max_exchanges=3)

    for turn in range(1, MAX_ITERATIONS +1):
        # Prints the prompt the model sees at the begining of each turn if Trace = True
        show(f"=== Turn {turn} ===\nPrompt:", prompt)

        reply = model(prompt)

        # Prints the models reply every turn if Trace = True
        show("Model reply:", reply)

        # If model attempts a final answer, this checks it against the answer key.
        # The shape of the answer is validated before accepting it.
        try:
            parsed = json.loads(reply)
        except json.JSONDecodeError:
            parsed = None
        if isinstance(parsed, dict) and "answer" in parsed:
            if isinstance(parsed["answer"], str) and isinstance(parsed.get("citations"), list):
                return parsed
            obs = ("Observation: invalid final answer: it must be a JSON object "
                    "with an 'answer' string and a 'citations' list.")
            # Otherwise it must be a tool call. VALIDATE BEFORE ANYTHING RUNS.
        else:
            verdict = validate_tool_call(reply)
            if not verdict["ok"]:
                # Rejected: the error becomes the Observation, the turn
                # repeats, and no tool ever saw the bad request.
                obs = error_observation(verdict)
            else:
                # Dispatch: the validated name picks the function, the
                # validated arguments unpack into its parameters.
                fn = TOOLS[verdict["tool"]]
                obs = observation_from(fn(**verdict["arguments"]))

        # Prints the observation if Trace = True
        show("Observation:", obs)

        # CHANGED (Week 3, Day 1): the window owns the history now. The exchange
        # goes into the window, and the prompt is rebuilt from whatever
        # the window still holds -- never from the full history again.
        window.append({"user": reply, "agent": obs})
        prompt = PROMPT.replace("{question}", question)
        for exchange in window.get():
            prompt += "\n" + exchange["user"] + "\n" + exchange["agent"] + "\n"

    return {"answer": "max iterations reached", "citations": []}


if __name__ == "__main__":
    result = run("List the files in my sandbox. Then read notes.txt and tell me what it says I do for a living and what I am learning. Then fetch the Python 3.13.0 release page on python.org and tell me what date Python 3.13.0 was released.")
    print("ANSWER:", result["answer"])
    print("CITATIONS:", result["citations"])

# Registry for the available tools.  
    # fetch_url: goes to a web page
    # run_command: safely runs a command in the SAFE_DIR
    # read_file: reads a file and returns it's context (either full or truncated).

    "fetch_url": fetch_url,
    "run_command": run_command,
    "read_file": read_file,

    fn = TOOLS.get(name.strip().lower())
    if fn is None:
        return f"unknown tool: {name}"
    return fn(arg)

Added in version 3.5. Standard Compliance and Interoperability ╢ The JSON format is specified by RFC 7159 and by ECMA-404 . This section details this moduleÆs level of compliance with the RFC. For simplicity, JSONEncoder and JSONDecoder subclasses, and parameters other than those explicitly mentioned, are not considered. This module does not comply with the RFC in a strict fashion, implementing some extensions that are valid JavaScript but not valid JSON. In particular: Infinite and NaN number values are accepted and output; Repeated names within an object are accepted, and only the value of the last name-value pair is used. Since the RFC permits RFC-compliant parsers to accept input texts that are not RFC-compliant, this moduleÆs deserializer is technically RFC-compliant under default settings. Character Encodings ╢ The RFC requires that JSON be represented using either UTF-8, UTF-16, or UTF-32, with UTF-8 being the recommended default for maximum interoperability. As permitted, though not required, by the RFC, this moduleÆs serializer sets ensure_ascii=True by default, thus escaping the output so that the resulting strings only contain printable ASCII characters. Other than the ensure_ascii parameter, this module is defined strictly in terms of conversion between Python objects and Unicode strings , and thus does not otherwise directly address the issue of character encodings. The RFC prohibits adding a byte order mark (BOM) to the start of a JSON text, and this moduleÆs serializer does not add a BOM to its output. The RFC permits, but does not require, JSON deserializers to ignore an initial BOM in their input. This moduleÆs deserializer raises a ValueError when an initial BOM is present. The RFC does not explicitly forbid JSON strings which contain byte sequences that donÆt correspond to valid Unicode characters (e.g. unpaired UTF-16 surrogates), but it does note that they may cause interoperability problems. By default, this module accepts and outputs (when present in the original str ) code points for such sequences. Infinite and NaN Number Values ╢ The RFC does not permit the representation of infinite or NaN number values. Despite that, by default, this module accepts and outputs Infinity , -Infinity , and NaN as if they were valid JSON number literal values: >>> # Neither of these calls raises an exception, but the results are not valid JSON >>> json . dumps ( float ( '-inf' )) '-Infinity' >>> json . dumps ( float ( 'nan' )) 'NaN' >>> # Same when deserializing >>> json . loads ( '-Infinity' ) -inf >>> json . loads ( 'NaN' ) nan In the serializer, the allow_nan parameter can be used to alter this behavior. In the deserializer, the parse_constant parameter can be used to alter this behavior. Repeated Names Within an Object ╢ The RFC specifies that the names within a JSON object should be unique, but does not mandate how repeated names in JSON objects should be handled. By default, this module does not raise an exception; instead, it ignores all but the last name-value pair for a given name: >>> weird_json = '{"x": 1, "x": 2, "x": 3}' >>> json . loads ( weird_json ) {'x': 3} The object_pairs_hook parameter can be used to alter this behavior. Top-level Non-Object, Non-Array Values ╢ The old version of JSON specified by the obsolete RFC 4627 required that the top-level value of a JSON text must be either a JSON object or array (Python dict or list ), and could not be a JSON null, boolean, number, or string value. RFC 7159 removed that restriction, and this module does not and has never implemented that restriction in either its serializer or its deserializer. Regardless, for maximum interoperability, you may wish to voluntarily adhere to the restriction yourself. Implementation Limitations ╢ Some JSON deserializer implementations may set limits on: the size of accepted JSON texts the maximum level of nesting of JSON objects and arrays the range and precision of JSON numbers the content and maximum length of JSON strings This module does not impose any such limits beyond those of the relevant Python datatypes themselves or the Python interpreter itself. def observation_from(result):
    """Wrap a tool's return dictionary as the Observation text. 
    Takes the tool's return and decide ok or error.
    """
    if result["ok"]:
        return f"Observation: {result.get('text') or result.get('stdout') or ''}"
    return f"Observation: tool error ({result['error']}): {result.get('detail', '')}"


def run(question, model=ask_gateway):
    """Runs through the whole loop.  
    - Ask the question.
    - Validate the response.
    - Dispatch a tool.
    - Append the observation.
    - Repeat until final answer.

    prompt = PROMPT.replace("{question}", question)

Added in version 3.5. Standard Compliance and Interoperability ╢ The JSON format is specified by RFC 7159 and by ECMA-404 . This section details this moduleÆs level of compliance with the RFC. For simplicity, JSONEncoder and JSONDecoder subclasses, and parameters other than those explicitly mentioned, are not considered. This module does not comply with the RFC in a strict fashion, implementing some extensions that are valid JavaScript but not valid JSON. In particular: Infinite and NaN number values are accepted and output; Repeated names within an object are accepted, and only the value of the last name-value pair is used. Since the RFC permits RFC-compliant parsers to accept input texts that are not RFC-compliant, this moduleÆs deserializer is technically RFC-compliant under default settings. Character Encodings ╢ The RFC requires that JSON be represented using either UTF-8, UTF-16, or UTF-32, with UTF-8 being the recommended default for maximum interoperability. As permitted, though not required, by the RFC, this moduleÆs serializer sets ensure_ascii=True by default, thus escaping the output so that the resulting strings only contain printable ASCII characters. Other than the ensure_ascii parameter, this module is defined strictly in terms of conversion between Python objects and Unicode strings , and thus does not otherwise directly address the issue of character encodings. The RFC prohibits adding a byte order mark (BOM) to the start of a JSON text, and this moduleÆs serializer does not add a BOM to its output. The RFC permits, but does not require, JSON deserializers to ignore an initial BOM in their input. This moduleÆs deserializer raises a ValueError when an initial BOM is present. The RFC does not explicitly forbid JSON strings which contain byte sequences that donÆt correspond to valid Unicode characters (e.g. unpaired UTF-16 surrogates), but it does note that they may cause interoperability problems. By default, this module accepts and outputs (when present in the original str ) code points for such sequences. Infinite and NaN Number Values ╢ The RFC does not permit the representation of infinite or NaN number values. Despite that, by default, this module accepts and outputs Infinity , -Infinity , and NaN as if they were valid JSON number literal values: >>> # Neither of these calls raises an exception, but the results are not valid JSON >>> json . dumps ( float ( '-inf' )) '-Infinity' >>> json . dumps ( float ( 'nan' )) 'NaN' >>> # Same when deserializing >>> json . loads ( '-Infinity' ) -inf >>> json . loads ( 'NaN' ) nan In the serializer, the allow_nan parameter can be used to alter this behavior. In the deserializer, the parse_constant parameter can be used to alter this behavior. Repeated Names Within an Object ╢ The RFC specifies that the names within a JSON object should be unique, but does not mandate how repeated names in JSON objects should be handled. By default, this module does not raise an exception; instead, it ignores all but the last name-value pair for a given name: >>> weird_json = '{"x": 1, "x": 2, "x": 3}' >>> json . loads ( weird_json ) {'x': 3} The object_pairs_hook parameter can be used to alter this behavior. Top-level Non-Object, Non-Array Values ╢ The old version of JSON specified by the obsolete RFC 4627 required that the top-level value of a JSON text must be either a JSON object or array (Python dict or list ), and could not be a JSON null, boolean, number, or string value. RFC 7159 removed that restriction, and this module does not and has never implemented that restriction in either its serializer or its deserializer. Regardless, for maximum interoperability, you may wish to voluntarily adhere to the restriction yourself. Implementation Limitations ╢ Some JSON deserializer implementations may set limits on: the size of accepted JSON texts the maximum level of nesting of JSON objects and arrays the range and precision of JSON numbers the content and maximum length of JSON strings This module does not impose any such limits beyond those of the relevant Python datatypes themselves or the Python interpreter itself. When serializing to JSON, beware any such limitations in applications that may consume your JSON. In particular, it is common for JSON numbers to be deserialized into IEEE 754 double precision numbers and thus subject to that representationÆs range and precision limitations. This is especially relevant when serializing Python int values of extremely large magnitude, or when serializing instances of ôexoticö numerical types such as decimal.Decimal . Observation:
Observation: """Day 3 - structured output: schemas as data, and the validator.

The model stops writing freeform Action: lines and must hand over JSON
that matches TOOL_SCHEMAS. Bad requests die here, named, before any
tool runs. Verdicts use the same {"ok": ...} shape as the tools.
"""

import json
from typing import Any

# ----------------------------------------------------------------------------
# The schemas, as data, not code. One entry per registered tool, written
# from the real Day 1/2 signatures. Each argument maps to (type, required?).
# JSON has one number type; Python splits int/float, so a "number" field
# accepts either -- and never True/False (bool is an int subclass in Python).
# ----------------------------------------------------------------------------

# road map for checking a respose.  Maps each tool to its rulebook,
# then maps argument names to a pair of facts about that argument.
TOOL_SCHEMAS: dict[str, dict[str, tuple[type, bool]]] = {
    "fetch_url": {"url": (str, True), "timeout": (float, False)},
    "run_command": {"command": (list, True), "timeout": (float, False)},
    "read_file": {"rel_path": (str, True)},
}

# define the allowable types
TYPE_NAMES = {str: "string", float: "number", list: "list"}

# checks if a value matches the type that the schema demands
# If schema wants a number, ints and floats are good, but never booleans.
# For everythiing else, do the plain type test.
def _type_matches(value: Any, expected: type) -> bool:
    if expected is float: # a JSON number: int or float, never a boolean
        return isinstance(value, (int,float)) and not isinstance(value, bool)
    return isinstance(value, expected)

# translates between python value type and the JSON name for that type.
def _type_name(value: Any) -> str:
    """The JSON-flavored name of a value's type, for error details."""
    if isinstance(value, bool):
        return "boolean"
    if isinstance(value, str):
        return "string"
    if isinstance(value, (int, float)):
        return "number"
    if isinstance(value, list):
        return "list"
    if isinstance(value, dict):
        return "object"
    return (type(value).__name__)

def validate_tool_call(raw_text: str) -> dict[str, Any]:
    """Check a raw model reply against TOOL_SCHEMAS.

    Returns {"ok": True, "tool": ..., "arguments": {...}} when the call
    may run, or {"ok": False, "error": ..., "detail": ...} when it may
    not. The detail names the exact field that failed: it is written to
    be pasted into the prompt for the retry, not just logged.
    """
    # Gate 1: is it JSON at all? Prose-wrapped JSON fails here on purpose.
    try:
        call = json.loads(raw_text)
    except(json.JSONDecodeError, TypeError):
        return {"ok": False,
                "error": "invalid_json",
                "detail": "reply was not valid JSON"}

    # Gate 2: is it an object carrying the two keys a call must have?
    if not isinstance(call, dict) or "tool" not in call or "arguments" not in call:
        return {"ok": False, "error": "invalid_json",
                "detail": "reply must be a JSON object with 'tool' and 'arguments'"}

    name, args = call["tool"], call["arguments"]

    # Gate 3: is the tool real? If not, say what is.
    if name not in TOOL_SCHEMAS:
        return {"ok": False, "error": "unknown_tool",
                "detail": f"unknown tool '{name}'. Available: {', '.join(TOOL_SCHEMAS)}"}
    if not isinstance(args, dict):
        return {"ok": False, "error": "bad_arguments",
                "detail": "'arguments' must be an object"}

    # Gate 4: field by field: required arguments present, types matching.
    for arg, (expected, required) in TOOL_SCHEMAS[name].items():
        if arg not in args:
            if required:
                return {"ok": False, "error": "bad_arguments",
                        "detail": f"arguments.{arg} is required"}
            continue
        if not _type_matches(args[arg], expected):
            return {"ok": False, "error": "bad_arguments",
                    "detail": f"arguments.{arg} must be a {TYPE_NAMES[expected]}, "
                              f"got a {_type_name(args[arg])}"}

    return {"ok": True, "tool": name, "arguments": args}


def error_observation(verdict: dict[str, Any]) -> str:
    """Turn a failed verdict into the Observation the model will read.
    Same mechanism as Week 1's format correction; the message is sharper
    because the validator knows exactly what failed."""
    return (f"Observation: invalid tool call ({verdict['error']}): {verdict['detail']}. "
            f"Reply with ONLY a JSON object: {{\"tool\": ..., \"arguments\": {{...}}}}.")

Added in version 3.5. Standard Compliance and Interoperability ╢ The JSON format is specified by RFC 7159 and by ECMA-404 . This section details this moduleÆs level of compliance with the RFC. For simplicity, JSONEncoder and JSONDecoder subclasses, and parameters other than those explicitly mentioned, are not considered. This module does not comply with the RFC in a strict fashion, implementing some extensions that are valid JavaScript but not valid JSON. In particular: Infinite and NaN number values are accepted and output; Repeated names within an object are accepted, and only the value of the last name-value pair is used. Since the RFC permits RFC-compliant parsers to accept input texts that are not RFC-compliant, this moduleÆs deserializer is technically RFC-compliant under default settings. Character Encodings ╢ The RFC requires that JSON be represented using either UTF-8, UTF-16, or UTF-32, with UTF-8 being the recommended default for maximum interoperability. As permitted, though not required, by the RFC, this moduleÆs serializer sets ensure_ascii=True by default, thus escaping the output so that the resulting strings only contain printable ASCII characters. Other than the ensure_ascii parameter, this module is defined strictly in terms of conversion between Python objects and Unicode strings , and thus does not otherwise directly address the issue of character encodings. The RFC prohibits adding a byte order mark (BOM) to the start of a JSON text, and this moduleÆs serializer does not add a BOM to its output. The RFC permits, but does not require, JSON deserializers to ignore an initial BOM in their input. This moduleÆs deserializer raises a ValueError when an initial BOM is present. The RFC does not explicitly forbid JSON strings which contain byte sequences that donÆt correspond to valid Unicode characters (e.g. unpaired UTF-16 surrogates), but it does note that they may cause interoperability problems. By default, this module accepts and outputs (when present in the original str ) code points for such sequences. Infinite and NaN Number Values ╢ The RFC does not permit the representation of infinite or NaN number values. Despite that, by default, this module accepts and outputs Infinity , -Infinity , and NaN as if they were valid JSON number literal values: >>> # Neither of these calls raises an exception, but the results are not valid JSON >>> json . dumps ( float ( '-inf' )) '-Infinity' >>> json . dumps ( float ( 'nan' )) 'NaN' >>> # Same when deserializing >>> json . loads ( '-Infinity' ) -inf >>> json . loads ( 'NaN' ) nan In the serializer, the allow_nan parameter can be used to alter this behavior. In the deserializer, the parse_constant parameter can be used to alter this behavior. Repeated Names Within an Object ╢ The RFC specifies that the names within a JSON object should be unique, but does not mandate how repeated names in JSON objects should be handled. By default, this module does not raise an exception; instead, it ignores all but the last name-value pair for a given name: >>> weird_json = '{"x": 1, "x": 2, "x": 3}' >>> json . loads ( weird_json ) {'x': 3} The object_pairs_hook parameter can be used to alter this behavior. # define the allowable types
TYPE_NAMES = {str: "string", float: "number", list: "list"}

# checks if a value matches the type that the schema demands
# If schema wants a number, ints and floats are good, but never booleans.
# For everythiing else, do the plain type test.
def _type_matches(value: Any, expected: type) -> bool:
    if expected is float: # a JSON number: int or float, never a boolean
        return isinstance(value, (int,float)) and not isinstance(value, bool)
    return isinstance(value, expected)

# translates between python value type and the JSON name for that type.
def _type_name(value: Any) -> str:
    """The JSON-flavored name of a value's type, for error details."""
    if isinstance(value, bool):
        return "boolean"
    if isinstance(value, str):
        return "string"
    if isinstance(value, (int, float)):
        return "number"
    if isinstance(value, list):
        return "list"
    if isinstance(value, dict):
        return "object"
    return (type(value).__name__)

def validate_tool_call(raw_text: str) -> dict[str, Any]:
    """Check a raw model reply against TOOL_SCHEMAS.

    Returns {"ok": True, "tool": ..., "arguments": {...}} when the call
    may run, or {"ok": False, "error": ..., "detail": ...} when it may
    not. The detail names the exact field that failed: it is written to
    be pasted into the prompt for the retry, not just logged.
    """
    # Gate 1: is it JSON at all? Prose-wrapped JSON fails here on purpose.
    try:
        call = json.loads(raw_text)
    except(json.JSONDecodeError, TypeError):
        return {"ok": False,
                "error": "invalid_json",
                "detail": "reply was not valid JSON"}

    # Gate 2: is it an object carrying the two keys a call must have?
    if not isinstance(call, dict) or "tool" not in call or "arguments" not in call:
        return {"ok": False, "error": "invalid_json",
                "detail": "reply must be a JSON object with 'tool' and 'arguments'"}

    name, args = call["tool"], call["arguments"]

    # Gate 3: is the tool real? If not, say what is.
    if name not in TOOL_SCHEMAS:
        return {"ok": False, "error": "unknown_tool",
                "detail": f"unknown tool '{name}'. Available: {', '.join(TOOL_SCHEMAS)}"}
    if not isinstance(args, dict):
        return {"ok": False, "error": "bad_arguments",
                "detail": "'arguments' must be an object"}

    # Gate 4: field by field: required arguments present, types matching.
    for arg, (expected, required) in TOOL_SCHEMAS[name].items():
        if arg not in args:
            if required:
                return {"ok": False, "error": "bad_arguments",
                        "detail": f"arguments.{arg} is required"}
            continue
        if not _type_matches(args[arg], expected):
            return {"ok": False, "error": "bad_arguments",
                    "detail": f"arguments.{arg} must be a {TYPE_NAMES[expected]}, "
                              f"got a {_type_name(args[arg])}"}

    return {"ok": True, "tool": name, "arguments": args}


def error_observation(verdict: dict[str, Any]) -> str:
    """Turn a failed verdict into the Observation the model will read.
    Same mechanism as Week 1's format correction; the message is sharper
    because the validator knows exactly what failed."""
    return (f"Observation: invalid tool call ({verdict['error']}): {verdict['detail']}. "
            f"Reply with ONLY a JSON object: {{\"tool\": ..., \"arguments\": {{...}}}}.")

def observation_from(result):
    """Wrap a tool's return dictionary as the Observation text. 
    Takes the tool's return and decide ok or error.
    """
    if result["ok"]:
        return f"Observation: {result.get('text') or result.get('stdout') or ''}"
    return f"Observation: tool error ({result['error']}): {result.get('detail', '')}"


def run(question, model=ask_gateway):
    """Runs through the whole loop.  
    - Ask the question.
    - Validate the response.
    - Dispatch a tool.
    - Append the observation.
    - Repeat until final answer.
    - Final answer dictionary:  {"answer": str, "citations": [str, ...]}."""

    prompt = PROMPT.replace("{question}", question)
    # NEW (Week 3, Day 1): the window owns this run's history
    window = SlidingWindow(max_exchanges=3)

    for turn in range(1, MAX_ITERATIONS +1):
        # Prints the prompt the model sees at the begining of each turn if Trace = True
        show(f"=== Turn {turn} ===\nPrompt:", prompt)

        reply = model(prompt)

        # Prints the models reply every turn if Trace = True
        show("Model reply:", reply)

        # If model attempts a final answer, this checks it against the answer key.
        # The shape of the answer is validated before accepting it.
        try:
            parsed = json.loads(reply)
        except json.JSONDecodeError:
            parsed = None
        if isinstance(parsed, dict) and "answer" in parsed:
            if isinstance(parsed["answer"], str) and isinstance(parsed.get("citations"), list):
                return parsed
            obs = ("Observation: invalid final answer: it must be a JSON object "
                    "with an 'answer' string and a 'citations' list.")
            # Otherwise it must be a tool call. VALIDATE BEFORE ANYTHING RUNS.
        else:
            verdict = validate_tool_call(reply)
            if not verdict["ok"]:
                # Rejected: the error becomes the Observation, the turn
                # repeats, and no tool ever saw the bad request.
                obs = error_observation(verdict)
            else:
                # Dispatch: the validated name picks the function, the
                # validated arguments unpack into its parameters.
                fn = TOOLS[verdict["tool"]]
                obs = observation_from(fn(**verdict["arguments"]))

        # Prints the observation if Trace = True
        show("Observation:", obs)

        # CHANGED (Week 3, Day 1): the window owns the history now. The exchange
        # goes into the window, and the prompt is rebuilt from whatever
        # the window still holds -- never from the full history again.
        window.append({"user": reply, "agent": obs})
        prompt = PROMPT.replace("{question}", question)
        for exchange in window.get():
            prompt += "\n" + exchange["user"] + "\n" + exchange["agent"] + "\n"

    return {"answer": "max iterations reached", "citations": []}

Added in version 3.5. Standard Compliance and Interoperability ╢ The JSON format is specified by RFC 7159 and by ECMA-404 . This section details this moduleÆs level of compliance with the RFC. For simplicity, JSONEncoder and JSONDecoder subclasses, and parameters other than those explicitly mentioned, are not considered. This module does not comply with the RFC in a strict fashion, implementing some extensions that are valid JavaScript but not valid JSON. In particular: Infinite and NaN number values are accepted and output; Repeated names within an object are accepted, and only the value of the last name-value pair is used. Since the RFC permits RFC-compliant parsers to accept input texts that are not RFC-compliant, this moduleÆs deserializer is technically RFC-compliant under default settings. Character Encodings ╢ The RFC requires that JSON be represented using either UTF-8, UTF-16, or UTF-32, with UTF-8 being the recommended default for maximum interoperability. As permitted, though not required, by the RFC, this moduleÆs serializer sets ensure_ascii=True by default, thus escaping the output so that the resulting strings only contain printable ASCII characters. Other than the ensure_ascii parameter, this module is defined strictly in terms of conversion between Python objects and Unicode strings , and thus does not otherwise directly address the issue of character encodings. The RFC prohibits adding a byte order mark (BOM) to the start of a JSON text, and this moduleÆs serializer does not add a BOM to its output. The RFC permits, but does not require, JSON deserializers to ignore an initial BOM in their input. This moduleÆs deserializer raises a ValueError when an initial BOM is present. The RFC does not explicitly forbid JSON strings which contain byte sequences that donÆt correspond to valid Unicode characters (e.g. unpaired UTF-16 surrogates), but it does note that they may cause interoperability problems. By default, this module accepts and outputs (when present in the original str ) code points for such sequences. Infinite and NaN Number Values ╢ The RFC does not permit the representation of infinite or NaN number values. Despite that, by default, this module accepts and outputs Infinity , -Infinity , and NaN as if they were valid JSON number literal values: >>> # Neither of these calls raises an exception, but the results are not valid JSON >>> json . dumps ( float ( '-inf' )) '-Infinity' >>> json . dumps ( float ( 'nan' )) 'NaN' >>> # Same when deserializing >>> json . loads ( '-Infinity' ) -inf >>> json . loads ( 'NaN' ) nan In the serializer, the allow_nan parameter can be used to alter this behavior. Repeated Names Within an Object ╢ The RFC specifies that the names within a JSON object should be unique, but does not mandate how repeated names in JSON objects should be handled. By default, this module does not raise an exception; instead, it ignores all but the last name-value pair for a given name: >>> weird_json = '{"x": 1, "x": 2, "x": 3}' >>> json . loads ( weird_json ) {'x': 3} The object_pairs_hook parameter can be used to alter this behavior. import json
from typing import Any

# ----------------------------------------------------------------------------
# The schemas, as data, not code. One entry per registered tool, written
# from the real Day 1/2 signatures. Each argument maps to (type, required?).
# JSON has one number type; Python splits int/float, so a "number" field
# accepts either -- and never True/False (bool is an int subclass in Python).
# ----------------------------------------------------------------------------

# road map for checking a respose.  Maps each tool to its rulebook,
# then maps argument names to a pair of facts about that argument.
TOOL_SCHEMAS: dict[str, dict[str, tuple[type, bool]]] = {
    "fetch_url": {"url": (str, True), "timeout": (float, False)},
    "run_command": {"command": (list, True), "timeout": (float, False)},
    "read_file": {"rel_path": (str, True)},
}

# define the allowable types
TYPE_NAMES = {str: "string", float: "number", list: "list"}

# checks if a value matches the type that the schema demands
# If schema wants a number, ints and floats are good, but never booleans.
# For everythiing else, do the plain type test.
def _type_matches(value: Any, expected: type) -> bool:
    if expected is float: # a JSON number: int or float, never a boolean
        return isinstance(value, (int,float)) and not isinstance(value, bool)
    return isinstance(value, expected)

# translates between python value type and the JSON name for that type.
def _type_name(value: Any) -> str:
    """The JSON-flavored name of a value's type, for error details."""
    if isinstance(value, bool):
        return "boolean"
    if isinstance(value, str):
        return "string"
    if isinstance(value, (int, float)):
        return "number"
    if isinstance(value, list):
        return "list"
    if isinstance(value, dict):
        return "object"
    return (type(value).__name__)

def validate_tool_call(raw_text: str) -> dict[str, Any]:
    """Check a raw model reply against TOOL_SCHEMAS.

    Returns {"ok": True, "tool": ..., "arguments": {...}} when the call
    may run, or {"ok": False, "error": ..., "detail": ...} when it may
    not. The detail names the exact field that failed: it is written to
    be pasted into the prompt for the retry, not just logged.
    """
    # Gate 1: is it JSON at all? Prose-wrapped JSON fails here on purpose.
    try:
        call = json.loads(raw_text)
    except(json.JSONDecodeError, TypeError):
        return {"ok": False,
                "error": "invalid_json",
                "detail": "reply was not valid JSON"}

    # Gate 2: is it an object carrying the two keys a call must have?
    if not isinstance(call, dict) or "tool" not in call or "arguments" not in call:
        return {"ok": False, "error": "invalid_json",
                "detail": "reply must be a JSON object with 'tool' and 'arguments'"}

    name, args = call["tool"], call["arguments"]

    # Gate 3: is the tool real? If not, say what is.
    if name not in TOOL_SCHEMAS:
        return {"ok": False, "error": "unknown_tool",
                "detail": f"unknown tool '{name}'. Available: {', '.join(TOOL_SCHEMAS)}"}
    if not isinstance(args, dict):
        return {"ok": False, "error": "bad_arguments",
                "detail": "'arguments' must be an object"}

    # Gate 4: field by field: required arguments present, types matching.
    for arg, (expected, required) in TOOL_SCHEMAS[name].items():
        if arg not in args:
            if required:
                return {"ok": False, "error": "bad_arguments",
                        "detail": f"arguments.{arg} is required"}
            continue
        if not _type_matches(args[arg], expected):
            return {"ok": False, "error": "bad_arguments",
                    "detail": f"arguments.{arg} must be a {TYPE_NAMES[expected]}, "
                              f"got a {_type_name(args[arg])}"}

    return {"ok": True, "tool": name, "arguments": args}


def error_observation(verdict: dict[str, Any]) -> str:
    """Turn a failed verdict into the Observation the model will read.
    Same mechanism as Week 1's format correction; the message is sharper
    because the validator knows exactly what failed."""
    return (f"Observation: invalid tool call ({verdict['error']}): {verdict['detail']}. "
            f"Reply with ONLY a JSON object: {{\"tool\": ..., \"arguments\": {{...}}}}.")

Question: {question}
"""


def observation_from(result):
    """Wrap a tool's return dictionary as the Observation text. 
    Takes the tool's return and decide ok or error.
    """
    if result["ok"]:
        return f"Observation: {result.get('text') or result.get('stdout') or ''}"
    return f"Observation: tool error ({result['error']}): {result.get('detail', '')}"


def run(question, model=ask_gateway):
    """Runs through the whole loop.  
    - Ask the question.
    - Validate the response.
    - Dispatch a tool.
    - Append the observation.
    - Repeat until final answer.
    - Final answer dictionary:  {"answer": str, "citations": [str, ...]}."""

    prompt = PROMPT.replace("{question}", question)
    # NEW (Week 3, Day 1): the window owns this run's history
    window = SlidingWindow(max_exchanges=3)

    for turn in range(1, MAX_ITERATIONS +1):
        # Prints the prompt the model sees at the begining of each turn if Trace = True
        show(f"=== Turn {turn} ===\nPrompt:", prompt)

        reply = model(prompt)

        # Prints the models reply every turn if Trace = True
        show("Model reply:", reply)

        # If model attempts a final answer, this checks it against the answer key.
        # The shape of the answer is validated before accepting it.
        try:
            parsed = json.loads(reply)
        except json.JSONDecodeError:
            parsed = None
        if isinstance(parsed, dict) and "answer" in parsed:
            if isinstance(parsed["answer"], str) and isinstance(parsed.get("citations"), list):
                return parsed
            obs = ("Observation: invalid final answer: it must be a JSON object "
                    "with an 'answer' string and a 'citations' list.")
            # Otherwise it must be a tool call. VALIDATE BEFORE ANYTHING RUNS.
        else:
            verdict = validate_tool_call(reply)
            if not verdict["ok"]:
                # Rejected: the error becomes the Observation, the turn
                # repeats, and no tool ever saw the bad request.
                obs = error_observation(verdict)
            else:
                # Dispatch: the validated name picks the function, the
                # validated arguments unpack into its parameters.
                fn = TOOLS[verdict["tool"]]
                obs = observation_from(fn(**verdict["arguments"]))

        # Prints the observation if Trace = True
        show("Observation:", obs)

        # CHANGED (Week 3, Day 1): the window owns the history now. The exchange
        # goes into the window, and the prompt is rebuilt from whatever
        # the window still holds -- never from the full history again.
        window.append({"user": reply, "agent": obs})
        prompt = PROMPT.replace("{question}", question)
        for exchange in window.get():
            prompt += "\n" + exchange["user"] + "\n" + exchange["agent"] + "\n"

    return {"answer": "max iterations reached", "citations": []}

Added in version 3.5. Standard Compliance and Interoperability ╢ The JSON format is specified by RFC 7159 and by ECMA-404 . This section details this moduleÆs level of compliance with the RFC. For simplicity, JSONEncoder and JSONDecoder subclasses, and parameters other than those explicitly mentioned, are not considered. This module does not comply with the RFC in a strict fashion, implementing some extensions that are valid JavaScript but not valid JSON. In particular: Infinite and NaN number values are accepted and output; Repeated names within an object are accepted, and only the value of the last name-value pair is used. Since the RFC permits RFC-compliant parsers to accept input texts that are not RFC-compliant, this moduleÆs deserializer is technically RFC-compliant under default settings. Character Encodings ╢ The RFC requires that JSON be represented using either UTF-8, UTF-16, or UTF-32, with UTF-8 being the recommended default for maximum interoperability. As permitted, though not required, by the RFC, this moduleÆs serializer sets ensure_ascii=True by default, thus escaping the output so that the resulting strings only contain printable ASCII characters. Other than the ensure_ascii parameter, this module is defined strictly in terms of conversion between Python objects and Unicode strings , and thus does not otherwise directly address the issue of character encodings. The RFC prohibits adding a byte order mark (BOM) to the start of a JSON text, and this moduleÆs serializer does not add a BOM to its output. The RFC permits, but does not require, JSON deserializers to ignore an initial BOM in their input. This moduleÆs deserializer raises a ValueError when an initial BOM is present. The RFC does not explicitly forbid JSON strings which contain byte sequences that donÆt correspond to valid Unicode characters (e.g. unpaired UTF-16 surrogates), but it does note that they may cause interoperability problems. By default, this module accepts and outputs (when present in the original str ) code points for such sequences. Infinite and NaN Number Values ╢ The RFC does not permit the representation of infinite or NaN number values. Despite that, by default, this module accepts and outputs Infinity , -Infinity , and NaN as if they were valid JSON number literal values: >>> # Neither of these calls raises an exception, but the results are not valid JSON >>> json . dumps ( float ( '-inf' )) '-Infinity' >>> json . dumps ( float ( 'nan' )) 'NaN' >>> # Same when deserializing >>> json . loads ( '-Infinity' ) -inf >>> json . loads ( 'NaN' ) nan In the serializer, the allow_nan parameter can be used to alter this behavior. In the deserializer, the parse_constant parameter can be used to alter this behavior. Repeated Names Within an Object ╢ The RFC specifies that the names within a JSON object should be unique, but does not mandate how repeated names in JSON objects should be handled. By default, this module does not raise an exception; instead, it ignores all but the last name-value pair for a given name: >>> weird_json = '{"x": 1, "x": 2, "x": 3}' >>> json . loads ( weird_json ) {'x': 3} The object_pairs_hook parameter can be used to alter this behavior. Top-level Non-Object, Non-Array Values ╢ The old version of JSON specified by the obsolete RFC 4627 required that the top-level value of a JSON text must be either a JSON object or array (Python dict or list ), and could not be a JSON null, boolean, number, or string value. Regardless, for maximum interoperability, you may wish to voluntarily adhere to the restriction yourself. TRACE = True # Set to False to run quietly. True prints the full trace every turn.

def show(label, text):
    """Print one labeled trace section, but only when TRACE is on."""
    if not TRACE:
        return
    print(label)
    print(text)
    print()

# The prompt is the contract, now shorter than week 1 as JSON needs less explaining.
# Also, the example is removed because JSON IS the example.
PROMPT = """You are a research agent. You answer questions by using tools, then cite your sources.
Reply with ONLY a JSON object, no other text. To use a tool:
{"tool": "<tool name>", "arguments": { ... }}
Available tools: run_command (arguments: command, a list), read_file (arguments: rel_path),
fetch_url (arguments: url, timeout).
Do not wrap your reply in markdown or code fences. Send the raw JSON object only.
Reply with exactly one JSON object and nothing else. Never write an Observation line. Observations come only from the tools, after your reply ends.
This computer runs Windows. There is no ls program. To list files, use run_command with ["cmd", "/c", "dir"].
When you have enough to answer, reply with ONLY:
{"answer": "<your answer>", "citations": ["<file or URL>", ...]}

Question: {question}
"""


def observation_from(result):
    """Wrap a tool's return dictionary as the Observation text. 
    Takes the tool's return and decide ok or error.
    """
    if result["ok"]:
        return f"Observation: {result.get('text') or result.get('stdout') or ''}"
    return f"Observation: tool error ({result['error']}): {result.get('detail', '')}"


def run(question, model=ask_gateway):
    """Runs through the whole loop.  
    - Ask the question.
    - Validate the response.
    - Dispatch a tool.
    - Append the observation.
    - Repeat until final answer.
    - Final answer dictionary:  {"answer": str, "citations": [str, ...]}."""

    prompt = PROMPT.replace("{question}", question)
    # NEW (Week 3, Day 1): the window owns this run's history
    window = SlidingWindow(max_exchanges=3)

    for turn in range(1, MAX_ITERATIONS +1):
        # Prints the prompt the model sees at the begining of each turn if Trace = True
        show(f"=== Turn {turn} ===\nPrompt:", prompt)

        reply = model(prompt)

        # Prints the models reply every turn if Trace = True
        show("Model reply:", reply)

        # If model attempts a final answer, this checks it against the answer key.
        # The shape of the answer is validated before accepting it.
        try:
            parsed = json.loads(reply)
        except json.JSONDecodeError:
            parsed = None
        if isinstance(parsed, dict) and "answer" in parsed:
            if isinstance(parsed["answer"], str) and isinstance(parsed.get("citations"), list):
                return parsed
            obs = ("Observation: invalid final answer: it must be a JSON object "
                    "with an 'answer' string and a 'citations' list.")
            # Otherwise it must be a tool call. VALIDATE BEFORE ANYTHING RUNS.
        else:
            verdict = validate_tool_call(reply)
            if not verdict["ok"]:
                # Rejected: the error becomes the Observation, the turn
                # repeats, and no tool ever saw the bad request.
                obs = error_observation(verdict)
            else:
                # Dispatch: the validated name picks the function, the
                # validated arguments unpack into its parameters.
                fn = TOOLS[verdict["tool"]]
                obs = observation_from(fn(**verdict["arguments"]))

        # Prints the observation if Trace = True
        show("Observation:", obs)

        # CHANGED (Week 3, Day 1): the window owns the history now. The exchange
        # goes into the window, and the prompt is rebuilt from whatever
        # the window still holds -- never from the full history again.
        window.append({"user": reply, "agent": obs})
        prompt = PROMPT.replace("{question}", question)
        for exchange in window.get():
            prompt += "\n" + exchange["user"] + "\n" + exchange["agent"] + "\n"

    return {"answer": "max iterations reached", "citations": []}

Added in version 3.5. Standard Compliance and Interoperability ╢ The JSON format is specified by RFC 7159 and by ECMA-404 . This section details this moduleÆs level of compliance with the RFC. For simplicity, JSONEncoder and JSONDecoder subclasses, and parameters other than those explicitly mentioned, are not considered. This module does not comply with the RFC in a strict fashion, implementing some extensions that are valid JavaScript but not valid JSON. In particular: Infinite and NaN number values are accepted and output; Repeated names within an object are accepted, and only the value of the last name-value pair is used. Since the RFC permits RFC-compliant parsers to accept input texts that are not RFC-compliant, this moduleÆs deserializer is technically RFC-compliant under default settings. Character Encodings ╢ The RFC requires that JSON be represented using either UTF-8, UTF-16, or UTF-32, with UTF-8 being the recommended default for maximum interoperability. As permitted, though not required, by the RFC, this moduleÆs serializer sets ensure_ascii=True by default, thus escaping the output so that the resulting strings only contain printable ASCII characters. Other than the ensure_ascii parameter, this module is defined strictly in terms of conversion between Python objects and Unicode strings , and thus does not otherwise directly address the issue of character encodings. The RFC prohibits adding a byte order mark (BOM) to the start of a JSON text, and this moduleÆs serializer does not add a BOM to its output. The RFC permits, but does not require, JSON deserializers to ignore an initial BOM in their input. This moduleÆs deserializer raises a ValueError when an initial BOM is present. The RFC does not explicitly forbid JSON strings which contain byte sequences that donÆt correspond to valid Unicode characters (e.g. unpaired UTF-16 surrogates), but it does note that they may cause interoperability problems. By default, this module accepts and outputs (when present in the original str ) code points for such sequences. Infinite and NaN Number Values ╢ The RFC does not permit the representation of infinite or NaN number values. Despite that, by default, this module accepts and outputs Infinity , -Infinity , and NaN as if they were valid JSON number literal values: >>> # Neither of these calls raises an exception, but the results are not valid JSON >>> json . dumps ( float ( '-inf' )) '-Infinity' >>> json . dumps ( float ( 'nan' )) 'NaN' >>> # Same when deserializing >>> json . loads ( '-Infinity' ) -inf >>> json . loads ( 'NaN' ) nan In the serializer, the allow_nan parameter can be used to alter this behavior. In the deserializer, the parse_constant parameter can be used to alter this behavior. Repeated Names Within an Object ╢ The RFC specifies that the names within a JSON object should be unique, but does not mandate how repeated names in JSON objects should be handled. By default, this module does not raise an exception; instead, it ignores all but the last name-value pair for a given name: >>> weird_json = '{"x": 1, "x": 2, "x": 3}' >>> json . loads ( weird_json ) {'x': 3} The object_pairs_hook parameter can be used to alter this behavior. Top-level Non-Object, Non-Array Values ╢ The old version of JSON specified by the obsolete RFC 4627 required that the top-level value of a JSON text must be either a JSON object or array (Python dict or list ), and could not be a JSON null, boolean, number, or string value. RFC 7159 removed that restriction, and this module does not and has never implemented that restriction in either its serializer or its deserializer. Regardless, for maximum interoperability, you may wish to voluntarily adhere to the restriction yourself. Implementation Limitations ╢ Some JSON deserializer implementations may set limits on: the size of accepted JSON texts the maximum level of nesting of JSON objects and arrays the range and precision of JSON numbers the content and maximum length of JSON strings This module does not impose any such limits beyond those of the relevant Python datatypes themselves or the Python interpreter itself. When serializing to JSON, beware any such limitations in applications that may consume your JSON. In particular, it is common for JSON numbers to be deserialized into IEEE 754 double precision numbers and thus subject to that representationÆs range and precision limitations. This is especially relevant when serializing Python int values of extremely large magnitude, or when serializing instances of ôexoticö numerical types such as decimal.Decimal . import json
from typing import Any

# ----------------------------------------------------------------------------
# The schemas, as data, not code. One entry per registered tool, written
# from the real Day 1/2 signatures. Each argument maps to (type, required?).
# JSON has one number type; Python splits int/float, so a "number" field
# accepts either -- and never True/False (bool is an int subclass in Python).
# ----------------------------------------------------------------------------

# road map for checking a respose.  Maps each tool to its rulebook,
# then maps argument names to a pair of facts about that argument.
TOOL_SCHEMAS: dict[str, dict[str, tuple[type, bool]]] = {
    "fetch_url": {"url": (str, True), "timeout": (float, False)},
    "run_command": {"command": (list, True), "timeout": (float, False)},
    "read_file": {"rel_path": (str, True)},
}

# define the allowable types
TYPE_NAMES = {str: "string", float: "number", list: "list"}

# checks if a value matches the type that the schema demands
# If schema wants a number, ints and floats are good, but never booleans.
# For everythiing else, do the plain type test.
def _type_matches(value: Any, expected: type) -> bool:
    if expected is float: # a JSON number: int or float, never a boolean
        return isinstance(value, (int,float)) and not isinstance(value, bool)
    return isinstance(value, expected)

# translates between python value type and the JSON name for that type.
def _type_name(value: Any) -> str:
    """The JSON-flavored name of a value's type, for error details."""
    if isinstance(value, bool):
        return "boolean"
    if isinstance(value, str):
        return "string"
    if isinstance(value, (int, float)):
        return "number"
    if isinstance(value, list):
        return "list"
    if isinstance(value, dict):
        return "object"
    return (type(value).__name__)

def validate_tool_call(raw_text: str) -> dict[str, Any]:
    """Check a raw model reply against TOOL_SCHEMAS.

    Returns {"ok": True, "tool": ..., "arguments": {...}} when the call
    may run, or {"ok": False, "error": ..., "detail": ...} when it may
    not. The detail names the exact field that failed: it is written to
    be pasted into the prompt for the retry, not just logged.
    """
    # Gate 1: is it JSON at all? Prose-wrapped JSON fails here on purpose.
    try:
        call = json.loads(raw_text)
    except(json.JSONDecodeError, TypeError):
        return {"ok": False,
                "error": "invalid_json",
                "detail": "reply was not valid JSON"}

    # Gate 2: is it an object carrying the two keys a call must have?
    if not isinstance(call, dict) or "tool" not in call or "arguments" not in call:
        return {"ok": False, "error": "invalid_json",
                "detail": "reply must be a JSON object with 'tool' and 'arguments'"}

    name, args = call["tool"], call["arguments"]

    # Gate 3: is the tool real? If not, say what is.
    if name not in TOOL_SCHEMAS:
        return {"ok": False, "error": "unknown_tool",
                "detail": f"unknown tool '{name}'. Available: {', '.join(TOOL_SCHEMAS)}"}
    if not isinstance(args, dict):
        return {"ok": False, "error": "bad_arguments",
                "detail": "'arguments' must be an object"}

    # Gate 4: field by field: required arguments present, types matching.
    for arg, (expected, required) in TOOL_SCHEMAS[name].items():
        if arg not in args:
            if required:
                return {"ok": False, "error": "bad_arguments",
                        "detail": f"arguments.{arg} is required"}
            continue
        if not _type_matches(args[arg], expected):
            return {"ok": False, "error": "bad_arguments",
                    "detail": f"arguments.{arg} must be a {TYPE_NAMES[expected]}, "
                              f"got a {_type_name(args[arg])}"}

    return {"ok": True, "tool": name, "arguments": args}

Added in version 3.5. Standard Compliance and Interoperability ╢ The JSON format is specified by RFC 7159 and by ECMA-404 . This section details this moduleÆs level of compliance with the RFC. For simplicity, JSONEncoder and JSONDecoder subclasses, and parameters other than those explicitly mentioned, are not considered. This module does not comply with the RFC in a strict fashion, implementing some extensions that are valid JavaScript but not valid JSON. In particular: Infinite and NaN number values are accepted and output; Repeated names within an object are accepted, and only the value of the last name-value pair is used. Since the RFC permits RFC-compliant parsers to accept input texts that are not RFC-compliant, this moduleÆs deserializer is technically RFC-compliant under default settings. Character Encodings ╢ The RFC requires that JSON be represented using either UTF-8, UTF-16, or UTF-32, with UTF-8 being the recommended default for maximum interoperability. As permitted, though not required, by the RFC, this moduleÆs serializer sets ensure_ascii=True by default, thus escaping the output so that the resulting strings only contain printable ASCII characters. Other than the ensure_ascii parameter, this module is defined strictly in terms of conversion between Python objects and Unicode strings , and thus does not otherwise directly address the issue of character encodings. The RFC prohibits adding a byte order mark (BOM) to the start of a JSON text, and this moduleÆs serializer does not add a BOM to its output. The RFC permits, but does not require, JSON deserializers to ignore an initial BOM in their input. This moduleÆs deserializer raises a ValueError when an initial BOM is present. The RFC does not explicitly forbid JSON strings which contain byte sequences that donÆt correspond to valid Unicode characters (e.g. unpaired UTF-16 surrogates), but it does note that they may cause interoperability problems. By default, this module accepts and outputs (when present in the original str ) code points for such sequences. Infinite and NaN Number Values ╢ The RFC does not permit the representation of infinite or NaN number values. Despite that, by default, this module accepts and outputs Infinity , -Infinity , and NaN as if they were valid JSON number literal values: >>> # Neither of these calls raises an exception, but the results are not valid JSON >>> json . dumps ( float ( '-inf' )) '-Infinity' >>> json . dumps ( float ( 'nan' )) 'NaN' >>> # Same when deserializing >>> json . loads ( '-Infinity' ) -inf >>> json . loads ( 'NaN' ) nan In the serializer, the allow_nan parameter can be used to alter this behavior. Repeated Names Within an Object ╢ The RFC specifies that the names within a JSON object should be unique, but does not mandate how repeated names in JSON objects should be handled. Observation: """Day 3 - structured output: schemas as data, and the validator.

The model stops writing freeform Action: lines and must hand over JSON
that matches TOOL_SCHEMAS. Bad requests die here, named, before any
tool runs. Verdicts use the same {"ok": ...} shape as the tools.
"""

import json
from typing import Any

# ----------------------------------------------------------------------------
# The schemas, as data, not code. One entry per registered tool, written
# from the real Day 1/2 signatures. Each argument maps to (type, required?).
# JSON has one number type; Python splits int/float, so a "number" field
# accepts either -- and never True/False (bool is an int subclass in Python).
# ----------------------------------------------------------------------------

# road map for checking a respose.  Maps each tool to its rulebook,
# then maps argument names to a pair of facts about that argument.
TOOL_SCHEMAS: dict[str, dict[str, tuple[type, bool]]] = {
    "fetch_url": {"url": (str, True), "timeout": (float, False)},
    "run_command": {"command": (list, True), "timeout": (float, False)},
    "read_file": {"rel_path": (str, True)},
}

# define the allowable types
TYPE_NAMES = {str: "string", float: "number", list: "list"}

# checks if a value matches the type that the schema demands
# If schema wants a number, ints and floats are good, but never booleans.
# For everythiing else, do the plain type test.
def _type_matches(value: Any, expected: type) -> bool:
    if expected is float: # a JSON number: int or float, never a boolean
        return isinstance(value, (int,float)) and not isinstance(value, bool)
    return isinstance(value, expected)

# translates between python value type and the JSON name for that type.
def _type_name(value: Any) -> str:
    """The JSON-flavored name of a value's type, for error details."""
    if isinstance(value, bool):
        return "boolean"
    if isinstance(value, str):
        return "string"
    if isinstance(value, (int, float)):
        return "number"
    if isinstance(value, list):
        return "list"
    if isinstance(value, dict):
        return "object"
    return (type(value).__name__)

def validate_tool_call(raw_text: str) -> dict[str, Any]:
    """Check a raw model reply against TOOL_SCHEMAS.

    Returns {"ok": True, "tool": ..., "arguments": {...}} when the call
    may run, or {"ok": False, "error": ..., "detail": ...} when it may
    not. The detail names the exact field that failed: it is written to
    be pasted into the prompt for the retry, not just logged.
    """
    # Gate 1: is it JSON at all? Prose-wrapped JSON fails here on purpose.
    try:
        call = json.loads(raw_text)
    except(json.JSONDecodeError, TypeError):
        return {"ok": False,
                "error": "invalid_json",
                "detail": "reply was not valid JSON"}

    # Gate 2: is it an object carrying the two keys a call must have?
    if not isinstance(call, dict) or "tool" not in call or "arguments" not in call:
        return {"ok": False, "error": "invalid_json",
                "detail": "reply must be a JSON object with 'tool' and 'arguments'"}

    name, args = call["tool"], call["arguments"]

    # Gate 3: is the tool real? If not, say what is.
    if name not in TOOL_SCHEMAS:
        return {"ok": False, "error": "unknown_tool",
                "detail": f"unknown tool '{name}'. Available: {', '.join(TOOL_SCHEMAS)}"}
    if not isinstance(args, dict):
        return {"ok": False, "error": "bad_arguments",
                "detail": "'arguments' must be an object"}

    # Gate 4: field by field: required arguments present, types matching.
    for arg, (expected, required) in TOOL_SCHEMAS[name].items():
        if arg not in args:
            if required:
                return {"ok": False, "error": "bad_arguments",
                        "detail": f"arguments.{arg} is required"}
            continue
        if not _type_matches(args[arg], expected):
            return {"ok": False, "error": "bad_arguments",
                    "detail": f"arguments.{arg} must be a {TYPE_NAMES[expected]}, "
                              f"got a {_type_name(args[arg])}"}

    return {"ok": True, "tool": name, "arguments": args}


def error_observation(verdict: dict[str, Any]) -> str:
    """Turn a failed verdict into the Observation the model will read.
    Same mechanism as Week 1's format correction; the message is sharper
    because the validator knows exactly what failed."""
    return (f"Observation: invalid tool call ({verdict['error']}): {verdict['detail']}. "
            f"Reply with ONLY a JSON object: {{\"tool\": ..., \"arguments\": {{...}}}}.")

def observation_from(result):
    """Wrap a tool's return dictionary as the Observation text. 
    Takes the tool's return and decide ok or error.
    """
    if result["ok"]:
        return f"Observation: {result.get('text') or result.get('stdout') or ''}"
    return f"Observation: tool error ({result['error']}): {result.get('detail', '')}"


def run(question, model=ask_gateway):
    """Runs through the whole loop.  
    - Ask the question.
    - Validate the response.
    - Dispatch a tool.
    - Append the observation.
    - Repeat until final answer.
    - Final answer dictionary:  {"answer": str, "citations": [str, ...]}."""

    prompt = PROMPT.replace("{question}", question)
    # NEW (Week 3, Day 1): the window owns this run's history
    window = SlidingWindow(max_exchanges=3)

    for turn in range(1, MAX_ITERATIONS +1):
        # Prints the prompt the model sees at the begining of each turn if Trace = True
        show(f"=== Turn {turn} ===\nPrompt:", prompt)

        reply = model(prompt)

        # Prints the models reply every turn if Trace = True
        show("Model reply:", reply)

        # If model attempts a final answer, this checks it against the answer key.
        # The shape of the answer is validated before accepting it.
        try:
            parsed = json.loads(reply)
        except json.JSONDecodeError:
            parsed = None
        if isinstance(parsed, dict) and "answer" in parsed:
            if isinstance(parsed["answer"], str) and isinstance(parsed.get("citations"), list):
                return parsed
            obs = ("Observation: invalid final answer: it must be a JSON object "
                    "with an 'answer' string and a 'citations' list.")
            # Otherwise it must be a tool call. VALIDATE BEFORE ANYTHING RUNS.
        else:
            verdict = validate_tool_call(reply)
            if not verdict["ok"]:
                # Rejected: the error becomes the Observation, the turn
                # repeats, and no tool ever saw the bad request.
                obs = error_observation(verdict)
            else:
                # Dispatch: the validated name picks the function, the
                # validated arguments unpack into its parameters.
                fn = TOOLS[verdict["tool"]]
                obs = observation_from(fn(**verdict["arguments"]))

        # Prints the observation if Trace = True
        show("Observation:", obs)

        # CHANGED (Week 3, Day 1): the window owns the history now. The exchange
        # goes into the window, and the prompt is rebuilt from whatever
        # the window still holds -- never from the full history again.
        window.append({"user": reply, "agent": obs})
        prompt = PROMPT.replace("{question}", question)
        for exchange in window.get():
            prompt += "\n" + exchange["user"] + "\n" + exchange["agent"] + "\n"

    return {"answer": "max iterations reached", "citations": []}

Added in version 3.5. Standard Compliance and Interoperability ╢ The JSON format is specified by RFC 7159 and by ECMA-404 . This section details this moduleÆs level of compliance with the RFC. For simplicity, JSONEncoder and JSONDecoder subclasses, and parameters other than those explicitly mentioned, are not considered. This module does not comply with the RFC in a strict fashion, implementing some extensions that are valid JavaScript but not valid JSON. In particular: Infinite and NaN number values are accepted and output; Repeated names within an object are accepted, and only the value of the last name-value pair is used. Since the RFC permits RFC-compliant parsers to accept input texts that are not RFC-compliant, this moduleÆs deserializer is technically RFC-compliant under default settings. Character Encodings ╢ The RFC requires that JSON be represented using either UTF-8, UTF-16, or UTF-32, with UTF-8 being the recommended default for maximum interoperability. As permitted, though not required, by the RFC, this moduleÆs serializer sets ensure_ascii=True by default, thus escaping the output so that the resulting strings only contain printable ASCII characters. Other than the ensure_ascii parameter, this module is defined strictly in terms of conversion between Python objects and Unicode strings , and thus does not otherwise directly address the issue of character encodings. The RFC prohibits adding a byte order mark (BOM) to the start of a JSON text, and this moduleÆs serializer does not add a BOM to its output. The RFC permits, but does not require, JSON deserializers to ignore an initial BOM in their input. This moduleÆs deserializer raises a ValueError when an initial BOM is present. The RFC does not explicitly forbid JSON strings which contain byte sequences that donÆt correspond to valid Unicode characters (e.g. unpaired UTF-16 surrogates), but it does note that they may cause interoperability problems. By default, this module accepts and outputs (when present in the original str ) code points for such sequences. Infinite and NaN Number Values ╢ The RFC does not permit the representation of infinite or NaN number values. Despite that, by default, this module accepts and outputs Infinity , -Infinity , and NaN as if they were valid JSON number literal values: >>> # Neither of these calls raises an exception, but the results are not valid JSON >>> json . dumps ( float ( '-inf' )) '-Infinity' >>> json . dumps ( float ( 'nan' )) 'NaN' >>> # Same when deserializing >>> json . loads ( '-Infinity' ) -inf >>> json . loads ( 'NaN' ) nan In the serializer, the allow_nan parameter can be used to alter this behavior. In the deserializer, the parse_constant parameter can be used to alter this behavior. Repeated Names Within an Object ╢ The RFC specifies that the names within a JSON object should be unique, but does not mandate how repeated names in JSON objects should be handled. By default, this module does not raise an exception; instead, it ignores all but the last name-value pair for a given name: >>> weird_json = '{"x": 1, "x": 2, "x": 3}' >>> json . loads ( weird_json ) {'x': 3} The object_pairs_hook parameter can be used to alter this behavior. Top-level Non-Object, Non-Array Values ╢ The old version of JSON specified by the obsolete RFC 4627 required that the top-level value of a JSON text must be either a JSON object or array (Python dict or list ), and could not be a JSON null, boolean, number, or string value. RFC 7159 removed that restriction, and this module does not and has never implemented that restriction in either its serializer or its deserializer. Regardless, for maximum interoperability, you may wish to voluntarily adhere to the restriction yourself. Implementation Limitations ╢ Some JSON deserializer implementations may set limits on: the size of accepted JSON texts the maximum level of nesting of JSON objects and arrays the range and precision of JSON numbers the content and maximum length of JSON strings This module does not impose any such limits beyond those of the relevant Python datatypes themselves or the Python interpreter itself. When serializing to JSON, beware any such limitations in applications that may consume your JSON. import json
from typing import Any

# ----------------------------------------------------------------------------
# The schemas, as data, not code. One entry per registered tool, written
# from the real Day 1/2 signatures. Each argument maps to (type, required?).
# JSON has one number type; Python splits int/float, so a "number" field
# accepts either -- and never True/False (bool is an int subclass in Python).
# ----------------------------------------------------------------------------

# road map for checking a respose.  Maps each tool to its rulebook,
# then maps argument names to a pair of facts about that argument.
TOOL_SCHEMAS: dict[str, dict[str, tuple[type, bool]]] = {
    "fetch_url": {"url": (str, True), "timeout": (float, False)},
    "run_command": {"command": (list, True), "timeout": (float, False)},
    "read_file": {"rel_path": (str, True)},
}

# define the allowable types
TYPE_NAMES = {str: "string", float: "number", list: "list"}

# checks if a value matches the type that the schema demands
# If schema wants a number, ints and floats are good, but never booleans.
# For everythiing else, do the plain type test.
def _type_matches(value: Any, expected: type) -> bool:
    if expected is float: # a JSON number: int or float, never a boolean
        return isinstance(value, (int,float)) and not isinstance(value, bool)
    return isinstance(value, expected)

# translates between python value type and the JSON name for that type.
def _type_name(value: Any) -> str:
    """The JSON-flavored name of a value's type, for error details."""
    if isinstance(value, bool):
        return "boolean"
    if isinstance(value, str):
        return "string"
    if isinstance(value, (int, float)):
        return "number"
    if isinstance(value, list):
        return "list"
    if isinstance(value, dict):
        return "object"
    return (type(value).__name__)

def validate_tool_call(raw_text: str) -> dict[str, Any]:
    """Check a raw model reply against TOOL_SCHEMAS.

    Returns {"ok": True, "tool": ..., "arguments": {...}} when the call
    may run, or {"ok": False, "error": ..., "detail": ...} when it may
    not. The detail names the exact field that failed: it is written to
    be pasted into the prompt for the retry, not just logged.
    """
    # Gate 1: is it JSON at all? Prose-wrapped JSON fails here on purpose.
    try:
        call = json.loads(raw_text)
    except(json.JSONDecodeError, TypeError):
        return {"ok": False,
                "error": "invalid_json",
                "detail": "reply was not valid JSON"}

    # Gate 2: is it an object carrying the two keys a call must have?
    if not isinstance(call, dict) or "tool" not in call or "arguments" not in call:
        return {"ok": False, "error": "invalid_json",
                "detail": "reply must be a JSON object with 'tool' and 'arguments'"}

    name, args = call["tool"], call["arguments"]

    # Gate 3: is the tool real? If not, say what is.
    if name not in TOOL_SCHEMAS:
        return {"ok": False, "error": "unknown_tool",
                "detail": f"unknown tool '{name}'. Available: {', '.join(TOOL_SCHEMAS)}"}
    if not isinstance(args, dict):
        return {"ok": False, "error": "bad_arguments",
                "detail": "'arguments' must be an object"}

    # Gate 4: field by field: required arguments present, types matching.
    for arg, (expected, required) in TOOL_SCHEMAS[name].items():
        if arg not in args:
            if required:
                return {"ok": False, "error": "bad_arguments",
                        "detail": f"arguments.{arg} is required"}
            continue
        if not _type_matches(args[arg], expected):
            return {"ok": False, "error": "bad_arguments",
                    "detail": f"arguments.{arg} must be a {TYPE_NAMES[expected]}, "
                              f"got a {_type_name(args[arg])}"}

    return {"ok": True, "tool": name, "arguments": args}


def error_observation(verdict: dict[str, Any]) -> str:
    """Turn a failed verdict into the Observation the model will read.
    Same mechanism as Week 1's format correction; the message is sharper
    because the validator knows exactly what failed."""
    return (f"Observation: invalid tool call ({verdict['error']}): {verdict['detail']}. "
            f"Reply with ONLY a JSON object: {{\"tool\": ..., \"arguments\": {{...}}}}.")

def observation_from(result):
    """Wrap a tool's return dictionary as the Observation text. 
    Takes the tool's return and decide ok or error.
    """
    if result["ok"]:
        return f"Observation: {result.get('text') or result.get('stdout') or ''}"
    return f"Observation: tool error ({result['error']}): {result.get('detail', '')}"


def run(question, model=ask_gateway):
    """Runs through the whole loop.  
    - Ask the question.
    - Validate the response.
    - Dispatch a tool.
    - Append the observation.
    - Repeat until final answer.
    - Final answer dictionary:  {"answer": str, "citations": [str, ...]}."""

    prompt = PROMPT.replace("{question}", question)
    # NEW (Week 3, Day 1): the window owns this run's history
    window = SlidingWindow(max_exchanges=3)

    for turn in range(1, MAX_ITERATIONS +1):
        # Prints the prompt the model sees at the begining of each turn if Trace = True
        show(f"=== Turn {turn} ===\nPrompt:", prompt)

        reply = model(prompt)

        # Prints the models reply every turn if Trace = True
        show("Model reply:", reply)

        # If model attempts a final answer, this checks it against the answer key.
        # The shape of the answer is validated before accepting it.
        try:
            parsed = json.loads(reply)
        except json.JSONDecodeError:
            parsed = None
        if isinstance(parsed, dict) and "answer" in parsed:
            if isinstance(parsed["answer"], str) and isinstance(parsed.get("citations"), list):
                return parsed
            obs = ("Observation: invalid final answer: it must be a JSON object "
                    "with an 'answer' string and a 'citations' list.")
            # Otherwise it must be a tool call. VALIDATE BEFORE ANYTHING RUNS.
        else:
            verdict = validate_tool_call(reply)
            if not verdict["ok"]:
                # Rejected: the error becomes the Observation, the turn
                # repeats, and no tool ever saw the bad request.
                obs = error_observation(verdict)
            else:
                # Dispatch: the validated name picks the function, the
                # validated arguments unpack into its parameters.
                fn = TOOLS[verdict["tool"]]
                obs = observation_from(fn(**verdict["arguments"]))

        # Prints the observation if Trace = True
        show("Observation:", obs)

        # CHANGED (Week 3, Day 1): the window owns the history now. The exchange
        # goes into the window, and the prompt is rebuilt from whatever
        # the window still holds -- never from the full history again.
        window.append({"user": reply, "agent": obs})
        prompt = PROMPT.replace("{question}", question)
        for exchange in window.get():
            prompt += "\n" + exchange["user"] + "\n" + exchange["agent"] + "\n"

    return {"answer": "max iterations reached", "citations": []}
